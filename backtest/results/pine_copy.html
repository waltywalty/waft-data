// Asia-open gold breakout - SIGNAL INDICATOR for manual execution through
// TradingView's native IBKR panel (which is manual-only; see README.md).
// Same detection logic as AsiaOpenGold_strategy.pine, no orders: it draws the
// range and filter state, fires alerts at entries and at the 16:00-NY exit,
// and shows a status card with everything needed to place the bracket by hand.
indicator("Asia Open Gold [signals]", overlay = true, max_lines_count = 100)
// v1.2: correlation-trend card row + REGIME ON/OFF/APPROACHING alerts

corrDays = input.int(20, "Correlation window, days", minval = 5)
corrMax  = input.float(0.50, "Trade only when correlation <=", step = 0.05)
stopMult = input.float(2.0, "Stop, multiples of range width", step = 0.5)
cardPos  = input.string("top right", "Status card position",
     options = ["top right", "top left", "bottom right", "bottom left"])
cardTxt  = input.string("large", "Status card text size", options = ["small", "normal", "large"])

TZ_HK  = "Asia/Hong_Kong"
TZ_LDN = "Europe/London"
TZ_NY  = "America/New_York"

// ---- correlation from CONFIRMED daily closes -------------------------------
gD = request.security(syminfo.tickerid, "D", close[1], lookahead = barmerge.lookahead_on)
aD = request.security(audSym,           "D", close[1], lookahead = barmerge.lookahead_on)
var gArr = array.new_float()
var aArr = array.new_float()
if (ta.change(gD) != 0 or ta.change(aD) != 0) and not na(gD) and not na(aD)
    array.push(gArr, gD)
    array.push(aArr, aD)
    if array.size(gArr) > corrDays + 1
        array.shift(gArr)
        array.shift(aArr)

if array.size(gArr) == corrDays + 1
    mg = 0.0
    ma = 0.0
    for i = 1 to corrDays
        mg += math.log(array.get(gArr, i) / array.get(gArr, i - 1))
        ma += math.log(array.get(aArr, i) / array.get(aArr, i - 1))
    mg /= corrDays
    ma /= corrDays
    sgg = 0.0
    saa = 0.0
    sga = 0.0
    for i = 1 to corrDays
        dg = math.log(array.get(gArr, i) / array.get(gArr, i - 1)) - mg
        da = math.log(array.get(aArr, i) / array.get(aArr, i - 1)) - ma
        sgg += dg * dg
        saa += da * da
        sga += dg * da
    corrVal := sgg > 0 and saa > 0 ? sga / math.sqrt(sgg * saa) : na
filterOk = not na(corrVal) and corrVal <= corrMax

// ---- regime-change detection (daily granularity) ---------------------------
var float corrPrev = na
var corrHist = array.new_float()

if newDailyCorr
    array.push(corrHist, corrVal)
    if array.size(corrHist) > 6
        array.shift(corrHist)
corr5ago  = array.size(corrHist) == 6 ? array.get(corrHist, 0) : na
regimeOn  = newDailyCorr and not na(corrPrev) and corrPrev > corrMax and corrVal <= corrMax
regimeOff = newDailyCorr and not na(corrPrev) and corrPrev <= corrMax and corrVal > corrMax
approach  = newDailyCorr and not na(corrPrev) and corrPrev > corrMax + 0.05 and
     corrVal <= corrMax + 0.05 and corrVal > corrMax

    corrPrev := corrVal

// ---- range and entry detection (identical to the strategy) ----------------
inRange    = not na(time(timeframe.period, "0930-1030", TZ_HK))
rangeStart = inRange and not inRange[1]
var float rHigh = na
var float rLow  = na
var int   rEndT = na
var int   tradeDay = na

    rLow  := low
if inRange
    rHigh := math.max(rHigh, high)
    rLow  := math.min(rLow, low)
if (not inRange) and inRange[1]
    rEndT := time_close[1]

cutoffT  = timestamp(TZ_LDN, year(time, TZ_LDN), month(time, TZ_LDN), dayofmonth(time, TZ_LDN), 8, 0)
blockEnd = not na(rEndT) and time_close > rEndT and (time_close - rEndT) % 3600000 == 0
armed    = barstate.isconfirmed and blockEnd and tradeDay != hkKey and
     time_close <= cutoffT and not na(rHigh) and rHigh > rLow and filterOk

longSig  = armed and close > rHigh
shortSig = armed and close < rLow
var int   sigCount = 0
var int   lastSigT = na
if longSig or shortSig
    tradeDay := hkKey
    sigCount += 1
    lastSigT := time

nyMins   = hour(time_close, TZ_NY) * 60 + minute(time_close, TZ_NY)
exitBell = barstate.isconfirmed and nyMins == 16 * 60

// ---- alerts (fire on bar close; attach webhooks or push notifications) -----
alertcondition(longSig,  "AsiaOpenGold LONG",  "Asia gold LONG at {{close}} - stop = entry - 2x range, flat 16:00 NY")
alertcondition(shortSig, "AsiaOpenGold SHORT", "Asia gold SHORT at {{close}} - stop = entry + 2x range, flat 16:00 NY")

bgcolor(inRange ? color.new(color.teal, 88) : na)
bgcolor(not filterOk and barstate.isconfirmed ? color.new(color.red, 94) : na)
plotshape(longSig,  "Long",  shape.triangleup,   location.belowbar, color.new(color.green, 0),
     size = size.normal, text = "LONG",  textcolor = color.new(color.green, 0))
plotshape(shortSig, "Short", shape.triangledown, location.abovebar, color.new(color.red, 0),
     size = size.normal, text = "SHORT", textcolor = color.new(color.red, 0))

pos = cardPos == "top right" ? position.top_right : cardPos == "top left" ? position.top_left :
     cardPos == "bottom right" ? position.bottom_right : position.bottom_left
tsz = cardTxt == "small" ? size.small : cardTxt == "normal" ? size.normal : size.large

Backtest strategy

The full rule for the Strategy Tester, with webhook-ready order alerts for a bridge later.

//@version=6
// Asia-open gold breakout, gated by the gold/AUDUSD correlation regime.
// Pine port of backtest/mt5/AsiaOpenGold.mq5 (research: backtest/README.md).
//
// Rule: range = first 60 min from 09:30 Asia/Hong_Kong; enter on the first
// 60-minute block to CLOSE beyond it; no entry at/after 08:00 Europe/London;
// trade only when the 20-day correlation of gold/AUDUSD daily log returns
// (closed daily bars only) is <= 0.50; stop 2x range; flat 16:00 America/New_York.
//
// Chart: XAUUSD (or MGC1!) on the 5-minute timeframe. Timeframes other than
// 5m will change block boundaries - keep 5m.
//
// IMPORTANT: TradingView's native IBKR panel is MANUAL trading only - this
// strategy cannot route orders to IBKR by itself. For automation, wire the
// order-fill alert (webhook JSON below) to a bridge service. See README.md.
strategy("Asia Open Gold [research port]", overlay = true,
     process_orders_on_close = true, calc_on_every_tick = false,
     initial_capital = 2000, default_qty_type = strategy.fixed, default_qty_value = 1,
     commission_type = strategy.commission.cash_per_order, commission_value = 0.15)

// ---- inputs ----------------------------------------------------------------
audSym    = input.symbol("OANDA:AUDUSD", "AUDUSD symbol (match your data feed)")
corrDays  = input.int(20,   "Correlation window, days", minval = 5)
corrMax   = input.float(0.50, "Trade only when correlation <=", step = 0.05)
useFilter = input.bool(true, "Apply the correlation filter")
stopMult  = input.float(2.0, "Stop, multiples of range width", step = 0.5)
riskPct   = input.float(1.0, "Risk per trade, % of equity", step = 0.25)
sizeMode  = input.string("risk %", "Position sizing", options = ["risk %", "fixed qty"])
fixedQty  = input.float(1.0, "Fixed qty (oz or contracts)", step = 0.1)
ptValue   = input.float(1.0, "Point value per unit ($/pt; XAUUSD oz=1, MGC=10)")

TZ_HK  = "Asia/Hong_Kong"
TZ_LDN = "Europe/London"
TZ_NY  = "America/New_York"

// ---- correlation from CONFIRMED daily closes (yesterday and older only) ----
// close[1] with lookahead_on is the standard non-repainting idiom: on every
// intraday bar it returns the last COMPLETED daily close.
gD = request.security(syminfo.tickerid, "D", close[1], lookahead = barmerge.lookahead_on)
aD = request.security(audSym,           "D", close[1], lookahead = barmerge.lookahead_on)

var gArr = array.new_float()
var aArr = array.new_float()
newDaily = ta.change(gD) != 0 or ta.change(aD) != 0
if newDaily and not na(gD) and not na(aD)
    array.push(gArr, gD)
    array.push(aArr, aD)
    if array.size(gArr) > corrDays + 1
        array.shift(gArr)
        array.shift(aArr)

corrVal = float(na)
if array.size(gArr) == corrDays + 1
    mg = 0.0
    ma = 0.0
    for i = 1 to corrDays
        mg += math.log(array.get(gArr, i) / array.get(gArr, i - 1))
        ma += math.log(array.get(aArr, i) / array.get(aArr, i - 1))
    mg /= corrDays
    ma /= corrDays
    sgg = 0.0
    saa = 0.0
    sga = 0.0
    for i = 1 to corrDays
        dg = math.log(array.get(gArr, i) / array.get(gArr, i - 1)) - mg
        da = math.log(array.get(aArr, i) / array.get(aArr, i - 1)) - ma
        sgg += dg * dg
        saa += da * da
        sga += dg * da
    corrVal := sgg > 0 and saa > 0 ? sga / math.sqrt(sgg * saa) : na
// data unusable -> SKIP the day, never trade unfiltered
filterOk = not useFilter or (not na(corrVal) and corrVal <= corrMax)

// ---- the opening range: 09:30-10:30 HKT ------------------------------------
inRange   = not na(time(timeframe.period, "0930-1030", TZ_HK))
rangeStart = inRange and not inRange[1]
rangeEnd   = (not inRange) and inRange[1]

var float rHigh = na
var float rLow  = na
var int   rEndT = na      // ms close-time of the range window
var int   tradeDay = na   // HK date key of the day already traded / attempted
if rangeStart
    rHigh := high
    rLow  := low
if inRange
    rHigh := math.max(rHigh, high)
    rLow  := math.min(rLow, low)
if rangeEnd
    rEndT := time_close[1]

// ---- entry: first 60m block (anchored on the range end) to close beyond ----
hkKey    = dayofmonth(time, TZ_HK) + 100 * month(time, TZ_HK) + 10000 * year(time, TZ_HK)
cutoffT  = timestamp(TZ_LDN, year(time, TZ_LDN), month(time, TZ_LDN), dayofmonth(time, TZ_LDN), 8, 0)
blockEnd = not na(rEndT) and time_close > rEndT and (time_close - rEndT) % 3600000 == 0

canTrade = barstate.isconfirmed and blockEnd and strategy.position_size == 0 and
     tradeDay != hkKey and time_close <= cutoffT and not na(rHigh) and rHigh > rLow and filterOk

rangeW = rHigh - rLow
qty(riskDist) =>
    sizeMode == "fixed qty" ? fixedQty :
     math.max(0.0, strategy.equity * riskPct / 100.0 / math.max(riskDist * ptValue, syminfo.mintick))

if canTrade and close > rHigh
    tradeDay := hkKey
    strategy.entry("L", strategy.long, qty = qty(stopMult * rangeW),
         alert_message = '{"action":"buy","symbol":"{{ticker}}","price":{{close}},"stop":' + str.tostring(close - stopMult * rangeW, "#.##") + ',"strategy":"AsiaOpenGold"}')
    strategy.exit("Lx", "L", stop = close - stopMult * rangeW,
         alert_message = '{"action":"exit","symbol":"{{ticker}}","strategy":"AsiaOpenGold"}')
if canTrade and close < rLow
    tradeDay := hkKey
    strategy.entry("S", strategy.short, qty = qty(stopMult * rangeW),
         alert_message = '{"action":"sell","symbol":"{{ticker}}","price":{{close}},"stop":' + str.tostring(close + stopMult * rangeW, "#.##") + ',"strategy":"AsiaOpenGold"}')
    strategy.exit("Sx", "S", stop = close + stopMult * rangeW,
         alert_message = '{"action":"exit","symbol":"{{ticker}}","strategy":"AsiaOpenGold"}')

// ---- time exit: 16:00 New York, DST handled by the timezone ---------------
nyMins = hour(time_close, TZ_NY) * 60 + minute(time_close, TZ_NY)
if strategy.position_size != 0 and nyMins >= 16 * 60
    strategy.close_all(comment = "16:00 NY",
         alert_message = '{"action":"flat","symbol":"{{ticker}}","strategy":"AsiaOpenGold"}')

// ---- visuals ---------------------------------------------------------------
plot(inRange or na(rHigh) ? na : rHigh, "Range high", color.new(color.teal, 0), 2, plot.style_linebr)
plot(inRange or na(rLow)  ? na : rLow,  "Range low",  color.new(color.teal, 0), 2, plot.style_linebr)
bgcolor(inRange ? color.new(color.teal, 88) : na)
bgcolor(not filterOk and barstate.isconfirmed ? color.new(color.red, 94) : na)
plot(corrVal, "corr(gold,AUD,20d)", display = display.data_window)

HSI pre-open fade (paper only)

Watch-list item: fades a >=0.3-ATR push in the 09:15-09:30 HKT futures-only window. Load on an OANDA HK33HKD 15-minute chart; alerts FADE LONG / FADE SHORT / EXIT. Parameters frozen until 80 trades.

//@version=6
// HSI PRE-OPEN FADE - watch-list item #3, PAPER ONLY (n=43 in the backtest;
// re-test bar is 80+ trades; parameters are FROZEN - do not tune).
// Rule: the 09:15-09:30 HKT bar (futures open before the cash open) moves
// >= 0.3 x ATR14 -> fade it at the 09:30 cash open; stop 0.5 x that bar's
// range beyond its extreme; flat 16:00 HKT.
// Chart: OANDA HK33HKD (or an equivalent HSI CFD with the 09:15 session
// step), 15-minute timeframe - the pre-open window is exactly one bar there.
indicator("HSI Pre-Open Fade [paper]", overlay = true)

trigATR = input.float(0.3, "Trigger, |push| as fraction of ATR14 (FROZEN)", step = 0.05)
stopK   = input.float(0.5, "Stop, fraction of pre-open range (FROZEN)", step = 0.25)
cardPos = input.string("top right", "Card position",
     options = ["top right", "top left", "bottom right", "bottom left"])

TZ_HK = "Asia/Hong_Kong"

// prior DAY range, confirmed bars only (non-repainting), 14-day average.
// Note: daily CFD bars include after-hours - a slightly wider ATR than the
// backtest's day-session ATR, i.e. marginally FEWER triggers (conservative).
dayRng = request.security(syminfo.tickerid, "D", (high - low)[1],
     lookahead = barmerge.lookahead_on)
var rArr = array.new_float()
if ta.change(dayRng) != 0 and not na(dayRng)
    array.push(rArr, dayRng)
    if array.size(rArr) > 14
        array.shift(rArr)
atr14 = array.size(rArr) == 14 ? array.avg(rArr) : na

inPre = not na(time(timeframe.period, "0915-0930", TZ_HK))
var float push  = na
var float preHi = na
var float preLo = na
var float stopL = na
var int   dir   = 0          // +1 fade-long (push was down), -1 fade-short
var int   sigDay = na
var int   sigCount = 0
dayKey = dayofmonth(time, TZ_HK) + 100 * month(time, TZ_HK) + 10000 * year(time, TZ_HK)

if inPre and barstate.isconfirmed
    push  := close - open
    preHi := high
    preLo := low
    if not na(atr14) and math.abs(push) >= trigATR * atr14 and sigDay != dayKey
        sigDay := dayKey
        sigCount += 1
        dir  := push > 0 ? -1 : 1
        stopL := push > 0 ? preHi + stopK * (preHi - preLo) : preLo - stopK * (preHi - preLo)
    else
        dir := 0

fadeLong  = dir == 1  and sigDay == dayKey and inPre[0] and barstate.isconfirmed
fadeShort = dir == -1 and sigDay == dayKey and inPre[0] and barstate.isconfirmed
hkMins   = hour(time_close, TZ_HK) * 60 + minute(time_close, TZ_HK)
exitBell = barstate.isconfirmed and hkMins == 16 * 60

alertcondition(fadeLong,  "HSI Fade LONG",  "HSI pre-open pushed DOWN >=0.3 ATR - paper LONG at the 09:30 open, stop below, flat 16:00 HKT")
alertcondition(fadeShort, "HSI Fade SHORT", "HSI pre-open pushed UP >=0.3 ATR - paper SHORT at the 09:30 open, stop above, flat 16:00 HKT")
alertcondition(exitBell,  "HSI Fade EXIT",  "16:00 HKT - flatten any open HSI paper fade")

bgcolor(inPre ? color.new(color.orange, 85) : na)
plot(dir != 0 and sigDay == dayKey and hkMins <= 16 * 60 ? stopL : na,
     "Stop", color.new(color.red, 0), 2, plot.style_linebr)
plotshape(fadeLong,  "Fade long",  shape.triangleup,   location.belowbar,
     color.new(color.green, 0), size = size.normal, text = "FADE L",
     textcolor = color.new(color.green, 0))
plotshape(fadeShort, "Fade short", shape.triangledown, location.abovebar,
     color.new(color.red, 0), size = size.normal, text = "FADE S",
     textcolor = color.new(color.red, 0))

pos = cardPos == "top right" ? position.top_right : cardPos == "top left" ? position.top_left :
     cardPos == "bottom right" ? position.bottom_right : position.bottom_left
C_LBL = color.rgb(42, 49, 64)
C_OK  = color.rgb(27, 110, 85)
C_NO  = color.rgb(120, 128, 145)
var table card = table.new(pos, 2, 5, border_width = 1, border_color = color.rgb(120, 128, 145))
if barstate.islast
    armedToday = sigDay == dayKey
    table.cell(card, 0, 0, "ATR14 (daily)", text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 1, 0, na(atr14) ? "loading" : str.tostring(atr14, "#"), text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 0, 1, "trigger needs", text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 1, 1, na(atr14) ? "-" : str.tostring(trigATR * atr14, "#") + " pts", text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 0, 2, "last pre-open push", text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 1, 2, na(push) ? "none yet" : str.tostring(push, "#"), text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 0, 3, "today", text_size = size.large, text_color = color.white, bgcolor = C_LBL)
    table.cell(card, 1, 3, armedToday ? (dir == 1 ? "FADE LONG" : "FADE SHORT") : "no trigger",
         text_size = size.large, text_color = color.white, bgcolor = armedToday ? C_OK : C_NO)
